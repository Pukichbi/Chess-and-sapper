/* Расстановка мин «вразброс»: из нескольких случайных кандидатов берём клетку, которая закрывает больше
   клеток без мин рядом (пустых). Так пустых полей получается чуть меньше. Цифры по-прежнему считаются
   по реально стоящим минам, поэтому они всегда совпадают с ними. SPREAD_TRIES больше = пустых меньше. */
const SPREAD_TRIES=3;
function spreadPick(avail,M,count){
  const cover=grid(0);
  for(let r=0;r<8;r++)for(let c=0;c<8;c++)if(M[r][c])for(const[a,b]of nbrs(r,c))cover[a][b]++;
  const pool=shuffle(avail.slice());
  for(let i=0;i<count&&pool.length;i++){
    let bi=0,bs=-1;
    const tries=Math.min(SPREAD_TRIES,pool.length);
    for(let t=0;t<tries;t++){
      const q=pool[t];let sc=0;
      for(const[a,b]of nbrs(q>>3,q&7))if(cover[a][b]===0&&!M[a][b])sc++;
      if(sc>bs){bs=sc;bi=t;}
    }
    const q=pool.splice(bi,1)[0];
    M[q>>3][q&7]=true;
    for(const[a,b]of nbrs(q>>3,q&7))cover[a][b]++;
  }
}

function solveLayout(zone){

      let ok=true;

      for(const ci of varCons[v]){const cn=cons[ci];if(cn.sum>cn.n||cn.sum+cn.rem<cn.n){ok=false;break;}}
      if(ok){val[v]=x;if(dfs(i+1))return true;}
      for(const ci of varCons[v]){const cn=cons[ci];cn.sum-=x;cn.rem++;}
    }

/* Автоматическое раскрытие безопасной области, как в «Сапёре».

function revealSafeArea(r,c){
  if(!G||G.mine[r][c]||G.crater[r][c])return;
  const queue=[[r,c]],seen=new Set();
  while(queue.length){
    const [a,b]=queue.shift(),key=a*8+b;
    if(seen.has(key)||!inB(a,b))continue;
    seen.add(key);
    if(G.mine[a][b]||G.crater[a][b])continue;
    if((a!==r||b!==c)&&G.board[a][b])continue;
    G.revealed[a][b]=true;
    if(G.num[a][b]===0){
      for(const [nr,nc] of nbrs(a,b)){
        const nk=nr*8+nc;
        if(!seen.has(nk)&&!G.mine[nr][nc]&&!G.crater[nr][nc]&&!G.board[nr][nc]){
          queue.push([nr,nc]);
        }else if(!seen.has(nk)&&!G.mine[nr][nc]&&!G.crater[nr][nc]&&G.num[nr][nc]>0&&!G.board[nr][nc]){
          queue.push([nr,nc]);
        }
      }
    }
  }
}

function applyMove(m,promoT){
  if(G.over)return false;
  const B=G.board,color=G.turn;
  const left=remain(color);
  if(left<=0)return false;
  G.clock[color]=left;G.turnAt=Date.now();
  const p=B[m.fr][m.fc];
  const wasPawn=p.t==='p';
  const name=desc(p);
  let captured=B[m.tr][m.tc];
  if(m.ep){captured=B[m.fr][m.tc];B[m.fr][m.tc]=null;}
  B[m.fr][m.fc]=null;B[m.tr][m.tc]=p;p.m=true;
  let opener=null;
  let openName=name,openVerb=PN[p.t][1]==='f'?'подорвалась':'подорвался';
  if(p.t!=='k')opener={r:m.tr,c:m.tc};
  if(m.castle){
    const rr=m.fr;let rook;
    if(m.castle==='k'){rook=B[rr][7];B[rr][7]=null;B[rr][5]=rook;opener={r:rr,c:5};}
    else{rook=B[rr][0];B[rr][0]=null;B[rr][3]=rook;opener={r:rr,c:3};}
    rook.m=true;openName=desc(rook);openVerb='подорвалась';
  }
  if(m.promo&&promoT)p.t=promoT;
  let exploded=false;
  if(opener){
    if(!G.placed)setupWhiteZone(opener.r,opener.c);
    else if(color==='b'&&!G.bFirst)setupBlackZone(opener.r,opener.c);
    if(color==='b')G.bFirst=true;
    exploded=openCell(opener.r,opener.c);
    if(exploded)G.boom={r:opener.r,c:opener.c,t:Date.now()};
    else revealSafeArea(opener.r,opener.c);
  }
  fx={boom:exploded,cap:!!captured,check:false};
  let text;
  if(m.castle)text=(m.castle==='k'?'Короткая':'Длинная')+' рокировка.';
  else text=`${name} ${sq(m.fr,m.fc)}→${sq(m.tr,m.tc)}`+(captured&&!exploded?`, берёт: ${desc(captured).toLowerCase()}.`:'.');
  if(exploded){
    text+=` 💥 ${openName} ${openVerb} на мине (${sq(opener.r,opener.c)})!`;
    if(captured)text+=` Взятая фигура (${desc(captured).toLowerCase()}) тоже потеряна.`;
  }
  if(m.promo&&promoT&&!exploded)text+=` Превращение: ${PN[promoT][0]}.`;
  G.ep=(m.dbl&&B[m.tr][m.tc]===p)?{r:(m.fr+m.tr)/2,c:m.fc}:null;
  G.last={fr:m.fr,fc:m.fc,tr:m.tr,tc:m.tc};
  if(wasPawn||captured||exploded)G.half=0;else G.half++;
  if(captured&&captured.t==='k'){G.msg=text;setOver({winner:color,reason:'король захвачен'});return true;}
  G.turn=other(color);
  const key=posKey();G.reps[key]=(G.reps[key]||0)+1;
  const mv=legal(B,G.turn,G.ep);
  const kOpp=findKing(B,G.turn),kMe=findKing(B,color);
  const meAtt=kMe&&attacked(B,kMe.r,kMe.c,G.turn);
  const oppInCheck=kOpp&&attacked(B,kOpp.r,kOpp.c,color);
  fx.check=!!(oppInCheck||meAtt);
  if(!meAtt&&mv.length===0){
    G.msg=text;
    if(oppInCheck)setOver({winner:color,reason:'мат'});else setOver({draw:true,reason:'пат'});
    return true;
  }
  if(insufficient(B)){G.msg=text;setOver({draw:true,reason:'недостаточно материала'});return true;}
  if(G.half>=100){G.msg=text;setOver({draw:true,reason:'правило 50 ходов'});return true;}
  if(G.reps[key]>=3){G.msg=text;setOver({draw:true,reason:'троекратное повторение'});return true;}
  if(meAtt)text+=` ⚠️ Король ${color==='w'?'белых':'чёрных'} под боем: ${sideName(G.turn)} могут взять его!`;
  else if(oppInCheck)text+=' Шах!';
  G.msg=text;
  return true;
}

function seedHistory(){moveHistory=[cloneGameState()];historyIndex=0;updateHistoryButtons();}
function recordHistory(){
  if(historyRestore||!G)return;
  if(historyIndex<moveHistory.length-1)moveHistory=moveHistory.slice(0,historyIndex+1);
  const snap=cloneGameState();
  const prev=moveHistory[moveHistory.length-1];
  if(!prev||JSON.stringify(prev)!==JSON.stringify(snap)){moveHistory.push(snap);historyIndex=moveHistory.length-1;}
  updateHistoryButtons();
}
function restoreHistory(index){
  if(!G||index<0||index>=moveHistory.length)return;
  historyRestore=true;
  G=JSON.parse(JSON.stringify(moveHistory[index]));G.turnAt=Date.now();
  historyIndex=index;fx=null;
  ui.sel=null;ui.moves=[];ui.modal=false;ui.overShown=false;ui.botBusy=false;ui.anim=false;
  closeModal();refreshLegal();ui.viewFlip=wantFlip();
  $('#gameMenu').hidden=true;$('#reactionPanel').hidden=true;
  historyRestore=false;render();updateHistoryButtons();
  if(G.over)showOver();else maybeBot();
}
function updateHistoryButtons(){
  const undo=$('#undoBtn'),redo=$('#redoBtn');if(!undo||!redo)return;
  const online=G&&G.mode==='online';
  undo.disabled=online||!G||historyIndex<=0;
  redo.disabled=online||!G||historyIndex>=moveHistory.length-1;
  undo.title=online?'Недоступно в онлайне':'Назад';redo.title=online?'Недоступно в онлайне':'Вперёд';
}
function undoMove(){
  if(!G||G.mode==='online'||historyIndex<=0)return;
  let target=historyIndex-1;
  // In bot mode, undo the player's move and the bot reply as one action.
  if(G.mode==='bot'&&historyIndex>=2&&G.turn===G.human)target=historyIndex-2;
  restoreHistory(target);
}
function redoMove(){
  if(!G||G.mode==='online'||historyIndex>=moveHistory.length-1)return;
  let target=historyIndex+1;
  if(G.mode==='bot'&&moveHistory[target]&&moveHistory[target].turn===G.bot&&target+1<moveHistory.length)target++;
  restoreHistory(target);
}
function doMove(m,promoT){if(G.mode==='online'){onlineMove(m,promoT);return;}if(applyMove(m,promoT))commit();}

  refreshLegal();ui.sel=null;ui.moves=[];

  recordHistory();
  syncOrientation();
  if(G.over)showOver();
  maybeBot();
}

      setTimeout(()=>{b.style.transition='none';b.style.transform='';ui.viewFlip=wantFlip();ui.anim=false;render();},650);
    },450);
  }else{ui.viewFlip=w;render();}
}
function tick(){
  if(!G||G.over||G.mode==='bot')return; // В режиме против бота таймер не используется.
  if(G.mode==='online'){renderClocks();onlineTickMsg();return;} // В онлайне время ведёт сервер.
  const rem=remain(G.turn);
  if(rem<=0){
    const lose=G.turn,win=other(lose);
    G.clock[lose]=0;G.turnAt=Date.now();

      :{draw:true,reason:'время вышло, но у соперника недостаточно материала для мата'});
    commit();return;
  }
  renderClocks();
}

const PV={p:100,n:320,b:330,r:500,q:900,k:0};
const MATE=100000;
function evalPos(B){
  let s=0;
  for(let r=0;r<8;r++)for(let c=0;c<8;c++){
    const p=B[r][c];if(!p)continue;
    let v=PV[p.t];
    if(p.t==='p'){const adv=p.c==='w'?6-r:r-1;v+=adv*6;if(c>=3&&c<=4)v+=adv*3;}
    else if(p.t==='n'||p.t==='b'){v+=(7-(Math.abs(3.5-r)+Math.abs(3.5-c)))*4;}
    else if(p.t==='q'){v+=(7-(Math.abs(3.5-r)+Math.abs(3.5-c)));}
    s+=p.c==='w'?v:-v;
  }
  return s;
}
// риск мины для закрытых клеток: использует только то, что видят оба игрока (открытые клетки, цифры, кратеры)
function riskMap(){
  const P=grid(0);
  if(!G.placed)return P;
  let craters=0,unrev=0;
  for(let r=0;r<8;r++)for(let c=0;c<8;c++){if(G.crater[r][c])craters++;if(!G.revealed[r][c])unrev++;}
  const dens=unrev>0?Math.max(0,G.N-craters)/unrev:0;
  const acc=grid(0),cnt=grid(0),zero=grid(false),one=grid(false);
  for(let r=0;r<8;r++)for(let c=0;c<8;c++){
    if(!G.revealed[r][c])continue;
    const un=[];let cr=0;
    for(const[a,b]of nbrs(r,c)){if(!G.revealed[a][b])un.push([a,b]);else if(G.crater[a][b])cr++;}
    if(!un.length)continue;
    const rem=Math.max(0,G.num[r][c]-cr);
    const pr=Math.min(1,rem/un.length);
    for(const[a,b]of un){acc[a][b]+=pr;cnt[a][b]++;if(rem===0)zero[a][b]=true;else if(rem>=un.length)one[a][b]=true;}
  }
  for(let r=0;r<8;r++)for(let c=0;c<8;c++){
    if(G.revealed[r][c])continue;
    if(zero[r][c])P[r][c]=0;
    else if(one[r][c])P[r][c]=1;
    else if(cnt[r][c])P[r][c]=Math.min(.95,Math.max(.02,acc[r][c]/cnt[r][c]));
    else P[r][c]=dens;
  }
  return P;
}
function riskPen(B,m,ctx){
  const p=B[m.fr][m.fc];if(!p||p.t==='k')return 0;
  let r=m.tr,c=m.tc,v=PV[p.t];
  if(m.castle){r=m.fr;c=m.castle==='k'?5:3;v=PV.r;}
  if(G.revealed[r][c])return 0;
  return ctx.P[r][c]*v*ctx.W;
}
function simP(B,m,color){const S2=sim(B,m);if(m.promo)S2[m.tr][m.tc]={t:'q',c:color,m:true};return S2;}
function orderMoves(B,moves){
  const sc=m=>{const t=B[m.tr][m.tc],p=B[m.fr][m.fc];return (t?PV[t.t]*10-PV[p.t]:0)+(m.promo?800:0);};
  return moves.map(m=>({m,s:sc(m)})).sort((a,b)=>b.s-a.s).map(x=>x.m);
}
function nega(B,color,ep,depth,alpha,beta,ply,ctx){
  if(Date.now()>ctx.deadline){ctx.timeout=true;return 0;}
  if(depth===0){const e=evalPos(B);return color==='w'?e:-e;}
  const moves=legal(B,color,ep,true);
  if(!moves.length){const k=findKing(B,color);return (k&&attacked(B,k.r,k.c,other(color)))?-MATE+ply:0;}
  let best=-Infinity;
  for(const m of orderMoves(B,moves)){
    const t=B[m.tr][m.tc];
    let sc;
    if(t&&t.t==='k')sc=MATE-ply;
    else{
      const nep=m.dbl?{r:(m.fr+m.tr)/2,c:m.fc}:null;
      sc=-nega(simP(B,m,color),other(color),nep,depth-1,-beta,-alpha,ply+1,ctx)-riskPen(B,m,ctx);
      if(ctx.timeout)return 0;
    }
    if(sc>best)best=sc;
    if(best>alpha)alpha=best;
    if(alpha>=beta)break;
  }
  return best;
}
function botChoose(level){
  const B=G.board,color=G.turn;
  const moves=legal(B,color,G.ep);
  if(!moves.length)return null;
  const kc=moves.find(m=>{const t=B[m.tr][m.tc];return t&&t.t==='k';});
  if(kc)return kc;
  if(level==='easy'&&Math.random()<.4)return moves[Math.floor(Math.random()*moves.length)];
  const cfg={easy:{d:1,ms:300,W:.25,noise:140},medium:{d:2,ms:900,W:1,noise:25},hard:{d:5,ms:1400,W:1,noise:6}}[level];
  const ctx={deadline:Date.now()+cfg.ms,timeout:false,P:riskMap(),W:cfg.W};
  const root=orderMoves(B,moves);
  let scores=root.map(()=>0);
  for(let d=1;d<=cfg.d;d++){
    const cur=[];let done=true;
    ctx.timeout=false;
    for(const m of root){
      const nep=m.dbl?{r:(m.fr+m.tr)/2,c:m.fc}:null;
      const sc=-nega(simP(B,m,color),other(color),nep,d-1,-Infinity,Infinity,1,ctx)-riskPen(B,m,ctx);
      if(ctx.timeout){done=false;break;}
      cur.push(sc);
    }
    if(!done)break;
    scores=cur;
  }
  let bi=0,bs=-Infinity;
  root.forEach((m,i)=>{const s=scores[i]+(Math.random()*2-1)*cfg.noise;if(s>bs){bs=s;bi=i;}});
  return root[bi];
}
function maybeBot(){
  if(!G||G.mode!=='bot'||G.over||G.turn!==G.bot||ui.botBusy)return;
  ui.botBusy=true;
  setTimeout(()=>{
    ui.botBusy=false;
    if(!G||G.over||G.turn!==G.bot)return;
    let mv=null;
    try{mv=botChoose(G.level);}catch(e){mv=G.legal.length?G.legal[Math.floor(Math.random()*G.legal.length)]:null;}
    if(mv)doMove(mv,'q');
  },500+Math.random()*400);
}

function onCell(r,c){

const sleep=ms=>new Promise(r=>setTimeout(r,ms));
function apiBase(){
  const u=String(window.SERVER_URL||'').trim().replace(/\/+$/,'');
  if(u)return u;
  const native=!!(window.Capacitor&&window.Capacitor.isNativePlatform&&window.Capacitor.isNativePlatform());
  if(!native&&/^https?:$/.test(location.protocol))return location.origin;
  return '';
}
async function api(path,body){
  const base=apiBase();
  if(!base)throw new Error('Адрес сервера не настроен. Укажи его в файле server-config.js.');
  const ctl=new AbortController(),t=setTimeout(()=>ctl.abort(),20000);
  try{
    const r=await fetch(base+path,{method:'POST',headers:{'Content-Type':'text/plain'},body:JSON.stringify(body),signal:ctl.signal});
    return await r.json().catch(()=>({ok:false,error:'bad',message:'Сервер вернул некорректный ответ.'}));
  }catch(e){
    throw new Error(e&&e.name==='AbortError'?'Сервер не отвечает. Попробуй ещё раз.':'Нет связи с сервером. Проверь интернет.');
  }finally{clearTimeout(t);}
}

function netOpen(){
  netClose();
  const base=apiBase();if(!base||!NET.code)return;
  const es=new EventSource(base+'/api/stream?code='+encodeURIComponent(NET.code)+'&token='+encodeURIComponent(NET.token));
  NET.es=es;
  es.addEventListener('state',e=>{
    const was=NET.lost;NET.lost=false;
    try{onlineState(JSON.parse(e.data));}catch(x){console.error(x);}
    if(was&&G&&G.mode==='online')render();
  });
  es.addEventListener('reaction',e=>{try{showReaction(JSON.parse(e.data).emoji);}catch(x){}});
  es.onerror=()=>{
    NET.lost=true;if(G&&G.mode==='online'&&screenName==='game')onlineTickMsg();
    if(es.readyState===2){NET.es=null;netCheck();} // браузер сдался: проверяем сервер сами
  };
  if(!NET.pingT)NET.pingT=setInterval(()=>{ // держим бесплатный сервер «проснувшимся» и проверяем, что партия жива
    if(NET.code)fetch(apiBase()+'/api/ping',{method:'POST',headers:{'Content-Type':'text/plain'},body:JSON.stringify({code:NET.code,token:NET.token})}).catch(()=>{});
  },240000);
}
async function netCheck(){
  if(NET.checking||!NET.code)return;
  NET.checking=true;
  try{
    for(let i=0;i<30&&NET.code;i++){
      try{
        const r=await fetch(apiBase()+'/api/ping',{method:'POST',headers:{'Content-Type':'text/plain'},body:JSON.stringify({code:NET.code,token:NET.token})});
        if(r.status===404||r.status===403){onlineGone('Партия закрыта на сервере. Возможно, сервер перезапускался или соперник давно не отвечал.');return;}
        if(r.ok){netOpen();return;}
      }catch(e){}
      await sleep(2000);
    }
  }finally{NET.checking=false;}
}
